function detectColors(text: string, defaultPrimary = "#1e293b", defaultSecondary = "#0f172a") {
  const t = text.toLowerCase();
  if (t.includes("navy") || t.includes("blue") || t.includes("bleu") || t.includes("azure")) {
    return { primary: "#1e3a8a", secondary: "#172554", accent: "#3b82f6", name: "NAVY" };
  }
  if (t.includes("charcoal") || t.includes("graphite") || t.includes("anthracite")) {
    return { primary: "#334155", secondary: "#1e293b", accent: "#94a3b8", name: "CHARCOAL" };
  }
  if (t.includes("black") || t.includes("noir") || t.includes("onyx") || t.includes("nero")) {
    return { primary: "#18181b", secondary: "#09090b", accent: "#52525b", name: "BLACK" };
  }
  if (t.includes("grey") || t.includes("gray") || t.includes("silver") || t.includes("slate")) {
    return { primary: "#64748b", secondary: "#475569", accent: "#cbd5e1", name: "SILVER GREY" };
  }
  if (t.includes("white") || t.includes("blanc") || t.includes("ivory") || t.includes("cream")) {
    return { primary: "#f8fafc", secondary: "#e2e8f0", accent: "#cbd5e1", name: "IVORY WHITE" };
  }
  if (t.includes("gold") || t.includes("yellow gold") || t.includes("doré")) {
    return { primary: "#d97706", secondary: "#b45309", accent: "#fef08a", name: "GOLD" };
  }
  if (t.includes("rose gold") || t.includes("pink") || t.includes("rose")) {
    return { primary: "#fb7185", secondary: "#e11d48", accent: "#fecdd3", name: "ROSE" };
  }
  if (t.includes("brown") || t.includes("tan") || t.includes("cognac") || t.includes("espresso") || t.includes("havana")) {
    return { primary: "#78350f", secondary: "#451a03", accent: "#d97706", name: "COGNAC" };
  }
  if (t.includes("burgundy") || t.includes("bordeaux") || t.includes("wine") || t.includes("oxblood") || t.includes("red")) {
    return { primary: "#881337", secondary: "#4c0519", accent: "#f43f5e", name: "BURGUNDY" };
  }
  if (t.includes("green") || t.includes("olive") || t.includes("emerald") || t.includes("vert")) {
    return { primary: "#065f46", secondary: "#064e3b", accent: "#34d399", name: "EMERALD" };
  }
  return { primary: defaultPrimary, secondary: defaultSecondary, accent: "#94a3b8", name: "SIGNATURE" };
}
function escapeXml(unsafe: string): string {
  return unsafe.replace(/[<>&'"]/g, (c) => {
    switch (c) {
      case "<": return "&lt;";
      case ">": return "&gt;";
      case "&": return "&amp;";
      case "'": return "&apos;";
      case '"': return "&quot;";
      default: return c;
    }
  });
}
